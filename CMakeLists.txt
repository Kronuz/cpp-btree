cmake_minimum_required(VERSION 3.20)
project(cpp_btree LANGUAGES CXX)

add_library(cpp_btree INTERFACE)
add_library(cpp_btree::cpp_btree ALIAS cpp_btree)
target_include_directories(cpp_btree INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(cpp_btree INTERFACE cxx_std_17)

if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  include(CTest)
  if(BUILD_TESTING)
    function(btree_test_profile target standard diagnostic)
      target_link_libraries(${target} PRIVATE cpp_btree::cpp_btree)
      set_target_properties(${target} PROPERTIES
        CXX_STANDARD ${standard} CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
      target_compile_options(${target} PRIVATE ${diagnostic})
    endfunction()

    foreach(standard IN ITEMS 17 20)
      foreach(mode IN ITEMS release debug)
        if(mode STREQUAL "release")
          set(diagnostic -DNDEBUG)
        else()
          set(diagnostic -UNDEBUG)
        endif()
        set(layout layout_${standard}_${mode})
        add_executable(${layout} tests/node_layout.cc)
        btree_test_profile(${layout} ${standard} ${diagnostic})
        add_test(NAME ${layout} COMMAND ${layout})
      endforeach()

      foreach(modes IN ITEMS RR RD DR DD)
        if(modes MATCHES "^R")
          set(main_diagnostic -DNDEBUG)
        else()
          set(main_diagnostic -UNDEBUG)
        endif()
        if(modes MATCHES "R$")
          set(mutate_diagnostic -DNDEBUG)
        else()
          set(mutate_diagnostic -UNDEBUG)
        endif()
        set(stem mixed_${standard}_${modes})
        add_library(${stem}_main OBJECT tests/mixed_mode_main.cc)
        btree_test_profile(${stem}_main ${standard} ${main_diagnostic})
        add_library(${stem}_mutate OBJECT tests/mixed_mode_mutate.cc)
        btree_test_profile(${stem}_mutate ${standard} ${mutate_diagnostic})
        foreach(order IN ITEMS main_first mutate_first)
          if(order STREQUAL "main_first")
            add_executable(${stem}_${order}
              $<TARGET_OBJECTS:${stem}_main> $<TARGET_OBJECTS:${stem}_mutate>)
          else()
            add_executable(${stem}_${order}
              $<TARGET_OBJECTS:${stem}_mutate> $<TARGET_OBJECTS:${stem}_main>)
          endif()
          add_test(NAME ${stem}_${order} COMMAND ${stem}_${order})
        endforeach()
      endforeach()
    endforeach()
  endif()
endif()
